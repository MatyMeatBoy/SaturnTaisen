cmake_minimum_required(VERSION 3.28)
project(saturntaisen_ra_tests LANGUAGES C CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The tests for Windows credential storage require Windows")
endif()
set(SATRECOMP_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../local/deps/satrecomp"
    CACHE PATH "SDK with the SaturnTaisen patches applied")
include("${SATRECOMP_SOURCE_DIR}/cmake/sakura_ra.cmake")

enable_testing()
foreach(case IN ITEMS client journal profile transport host)
    set(test "tra_${case}")
    add_executable("${test}" "test_ra_${case}.cpp")
    target_include_directories("${test}" PRIVATE "${SATRECOMP_SOURCE_DIR}/port")
    target_compile_features("${test}" PRIVATE cxx_std_20)
    target_link_libraries("${test}" PRIVATE sakura_ra)
    if(MSVC)
        target_compile_options("${test}" PRIVATE /UNDEBUG)
    else()
        target_compile_options("${test}" PRIVATE -UNDEBUG)
    endif()
    if(case STREQUAL "transport")
        add_test(NAME "${test}" COMMAND "${test}")
    else()
        set(profile "${CMAKE_CURRENT_BINARY_DIR}/profile-${case}")
        add_test(NAME "${test}" COMMAND "${test}" "${profile}")
    endif()
endforeach()
